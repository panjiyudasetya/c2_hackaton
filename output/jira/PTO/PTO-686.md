Added a calculated column:



{noformat}Timecharter Estimation (Clarkson's Averages) = 
VAR BulkCarrierTCE = 
if(Ship[Dry Bulk Category]= "1) Mini bulk", 10000,
if(Ship[Dry Bulk Category]= "2) Handysize", 12000,
if(Ship[Dry Bulk Category]= "3) Supramax", 13000,
if(Ship[Dry Bulk Category]= "4) Panamax", 14000,
if(Ship[Dry Bulk Category]= "5) Cape", 15500,
if(Ship[Dry Bulk Category]= "6) Ultra Cape", 18000
))))))

VAR TankerTCE = 
if(Ship[Wet Bulk Category]= "1) BARGE", 18000,
if(Ship[Wet Bulk Category]= "2) GP", 26000,
if(Ship[Wet Bulk Category]= "3) MR", 30000,
if(Ship[Wet Bulk Category]= "4) LR1", 33000,
if(Ship[Wet Bulk Category]= "5) Aframax LR2", 42000,
if(Ship[Wet Bulk Category]= "6) LR2", 42000,
if(Ship[Wet Bulk Category]= "7) ULCC", 44000
)))))))

VAR ContainerTCE = 1

RETURN
IF(Ship[Ship Type] = "BULK_CARRIER", BulkCarrierTCE,
    IF(Ship[Ship Type] = "TANKER", TankerTCE,
        IF(Ship[Ship Type] = "CONTAINER", ContainerTCE)
    )
    ){noformat}
