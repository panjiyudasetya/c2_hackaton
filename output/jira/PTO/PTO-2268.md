### Panji Y. Wiwaha — 2025-10-20

Hey [~accountid:712020:305e4071-b713-4c7d-b893-28053a644e5e] 👋 

I’ve checked tugs activity based on the SOFs data that is successfylly ingested at October 15 without additional filtering (i.e.: {{ghost_ship_enable = false}}, {{ship_disabled = false}}, etc.) and found that the number of tug activities almost have no difference with LIVE data mart.

!image-20251020-095559.png|width=1800,alt="image-20251020-095559.png"!

This is the query that I ran:

{noformat}WITH sof_brssz AS (
	SELECT
		sof.generation_id AS port_visit_id
		, sof.entry_id AS visit_id
		, (sof."start" ->> 'time')::timestamp AS port_visit_start_ts
		, (sof."end" ->> 'time')::timestamp AS port_visit_end_ts
		, sof.berth_visits
	FROM stg_statement_of_fact sof
	WHERE sof.unlocode = 'BRSSZ'
)
, berth_visit AS (
	SELECT
		port_visit_id
		, port_visit_start_ts
		, port_visit_end_ts
		, visit_id
        , t.doc ->> 'id' AS berth_visit_id
		, (t.doc -> 'start' ->> 'time')::timestamp AS berth_visit_start_ts
		, (t.doc -> 'end' ->> 'time')::timestamp AS berth_visit_end_ts
        , jsonb_array_length(t.doc -> 'arrivalTugs') AS arrival_tugs_count
        , t.doc -> 'arrivalTugs' AS arrival_tugs
        , jsonb_array_length(t.doc -> 'departureTugs') AS departure_tugs_count
        , t.doc -> 'departureTugs' AS departure_tugs
	FROM sof_brssz, jsonb_array_elements("berth_visits") AS t (doc)
)
, aggregate_berth_visit AS (
	SELECT
		port_visit_start_ts
		, to_char(port_visit_start_ts, 'YYYY-MM') AS port_visit_month
		, berth_visit_start_ts
		, to_char(berth_visit_start_ts, 'YYYY-MM') AS berth_visit_month
		, arrival_tugs_count
		, departure_tugs_count
		, (arrival_tugs_count + departure_tugs_count) AS tugs_activity_count
	FROM berth_visit
)

SELECT * FROM aggregate_berth_visit
ORDER BY berth_visit_start_ts{noformat}
